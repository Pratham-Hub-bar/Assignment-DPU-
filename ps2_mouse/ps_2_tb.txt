module ps2_mouse_tb();
    reg clk, reset;
    reg [7:0] ps_in;
    wire done;
    wire [7:0] byte_1, byte_2, byte_3;

    ps2_mouse uut(.clk(clk), .reset(reset), .done(done), .ps_in(ps_in), .byte_1(byte_1), .byte_2(byte_2), .byte_3(byte_3)); 
    always #5 clk =~clk;

    initial begin
        clk = 0;
        ps_in = 8'b10101110;
        reset = 1'b1; #20;

        reset = 1'b0; ps_in = 8'b10100010; #10;
        ps_in = 8'b01100011; #10;

        ps_in = 8'b10101010; #10;
        ps_in = 8'd120; #10;
        ps_in = 8'd99; #10;

        ps_in = 8'b10000000; #10;
        ps_in = 8'b01111111; #10;
        ps_in = 8'd19; #10;
        ps_in = 8'd111; #10;

        reset = 1'b1; #20;
        $stop;
    end
endmodule
